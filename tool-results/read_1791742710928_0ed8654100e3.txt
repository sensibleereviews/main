   255→function QrKitCard({ locations, onPreviewGuest }: { locations: CampaignsPayload["locations"]; onPreviewGuest: () => void }) {
   256→  const { toast } = useToast();
   257→  const firstCode = locations[0]?.code;
   258→  const qr = useApiData<{ dataUrl: string; url: string }>(firstCode ? `/api/qr?code=${firstCode}` : null);
   259→
   260→  const copy = async () => {
   261→    if (!qr.data) return;
   262→    try {
   263→      await navigator.clipboard.writeText(qr.data.url);
   264→      toast({ title: "Link copied", description: qr.data.url });
   265→    } catch {
   266→      toast({ title: "Copy failed", description: qr.data.url });
   267→    }
   268→  };
   269→
   270→  const download = () => {
   271→    if (!qr.data) return;
   272→    const a = document.createElement("a");
   273→    a.href = qr.data.dataUrl;
   274→    a.download = `sensiblee-qr-${firstCode}.png`;
   275→    a.click();
   276→    toast({ title: "QR kit downloading", description: "Print-ready PNG · 480 px" });
   277→  };
   278→
   279→  return (
   280→    <ChartCard title="Table QR kit" subtitle="Module B · live scannable code">
   281→      <div className="flex items-center gap-4">
   282→        <div className="rounded-2xl border border-neutral-200 bg-cream p-2.5">
   283→          {qr.data ? (
   284→            <img src={qr.data.dataUrl} alt={`QR code for ${locations[0]?.name || "location"}`} width={124} height={124} className="rounded-lg" />
   285→          ) : (
   286→            <div className="h-[124px] w-[124px] animate-pulse rounded-lg bg-surface" />
   287→          )}
   288→        </div>
   289→        <div className="flex-1 text-[12.5px] leading-relaxed text-neutral-500">
   290→          <p className="font-semibold text-brand">scan → rate us in 2 taps</p>
   291→          <p className="mt-1">
   292→            Branded landing with your logo, star pre-select and Google hand-off. Printed at 40×40 mm on table tents.
   293→          </p>
   294→          <p className="mt-1 truncate font-mono text-[10.5px] text-neutral-400">{qr.data?.url || "…"}</p>
   295→          <div className="mt-3 flex flex-wrap gap-2">
   296→            <Button size="sm" variant="outline" className="h-8 text-xs" onClick={download} disabled={!qr.data}>
   297→              <Download size={13} className="mr-1" /> Kit
   298→            </Button>
   299→            <Button size="sm" variant="outline" className="h-8 text-xs" onClick={copy} disabled={!qr.data}>
   300→              <Copy size={13} className="mr-1" /> Link

   304→            </Button>
   305→          </div>
   306→        </div>
   307→      </div>
   308→    </ChartCard>
   309→  );

   314→type QueueItem = {
   315→  id: string; reviewStars: number; author: string; location: string; excerpt: string;
   316→  draft: string; tone: string; confidence: number; age: string; urgent?: boolean; text: string;
   317→};
   318→
   319→export function AiAssistantView() {
   320→  const { toast } = useToast();
   321→  const queue = useApiData<{ queue: QueueItem[] }>("/api/ai/queue");
   322→  const analytics = useApiData<{ kpis: { draftsPending: number; draftsApproved30d: number; atRisk: number; medianResponse: string } }>("/api/analytics");
   323→  const [drafts, setDrafts] = useState<Record<string, string>>({});
   324→  const [autoPilot, setAutoPilot] = useState(false);
   325→  const [busyId, setBusyId] = useState<string | null>(null);
   326→  const [approvedIds, setApprovedIds] = useState<string[]>([]);
   327→
   328→  const items = (queue.data?.queue || []).filter((i) => !approvedIds.includes(i.id));
   329→  const k = analytics.data?.kpis;
   330→
   331→  const bodyOf = (item: QueueItem) => drafts[item.id] ?? item.draft;
   332→
   333→  const approve = async (item: QueueItem) => {
   334→    setBusyId(item.id);
   335→    try {
   336→      await api.post(`/api/reviews/${item.id}/reply`, { body: bodyOf(item) });
   337→      setApprovedIds((a) => [...a, item.id]);
   338→      toast({ title: "Reply posted", description: `Published to Google for ${item.author}'s review.` });
   339→      queue.reload();
   340→    } catch (e) {
   341→      toast({ title: "Could not post", description: e instanceof Error ? e.message : "", variant: "destructive" });
   342→    } finally {
   343→      setBusyId(null);
   344→    }
   345→  };
   346→
   347→  const regenerate = async (item: QueueItem) => {
   348→    setBusyId(item.id);
   349→    try {
   350→      toast({ title: "Regenerating draft", description: "The AI worker is rewriting this reply in your voice." });
   351→      const res = await api.post<{ review: ReviewVM }>(`/api/reviews/${item.id}/draft`);
   352→      if (res.review.draft) setDrafts((d) => ({ ...d, [item.id]: res.review.draft! }));
   353→      queue.reload();
   354→    } catch (e) {
   355→      toast({ title: "Regeneration failed", description: e instanceof Error ? e.message : "", variant: "destructive" });
   356→    } finally {
   357→      setBusyId(null);
   358→    }
   359→  };
   360→
   361→  const toggleAutoPilot = async (v: boolean) => {
   362→    setAutoPilot(v);
   363→    try {
   364→      await api.patch("/api/settings", { ai: { threshold: v ? 95 : 90 } });
   365→      toast({
   366→        title: v ? "Auto-pilot staged" : "Auto-pilot off",
   367→        description: v
   368→          ? "Replies above 95% confidence would post automatically once you enable posting — 1–2★ reviews always wait for you."
   369→          : "Back to manual approval for all replies.",
   370→      });
   371→    } catch {
   372→      setAutoPilot(!v);
   373→      toast({ title: "Could not update", description: "Try again in a moment.", variant: "destructive" });
   374→    }
   375→  };
   376→
   377→  return (
   378→    <div className="space-y-5">
   379→      <div className="grid gap-5 lg:grid-cols-3">
   380→        <div className="lg:col-span-2 rounded-2xl border border-neutral-200/70 bg-white p-5">
   381→          <div className="flex flex-wrap items-center justify-between gap-3">
   382→            <div>
   383→              <h3 className="flex items-center gap-2 text-[15px] font-semibold text-brand">
   384→                <Sparkles size={16} className="text-mint-deep" /> Approval queue
   385→                <Badge className="bg-amber-soft text-amber-deep hover:bg-amber-soft">{items.length} pending</Badge>
   386→              </h3>
   387→              <p className="mt-0.5 text-xs text-neutral-400">Module C · every reply is yours to approve, edit or regenerate</p>
   388→            </div>
   389→            <div className="flex items-center gap-2.5 rounded-xl bg-surface px-3 py-2">
   390→              <Switch checked={autoPilot} onCheckedChange={toggleAutoPilot} />
   391→              <div className="text-[12px] leading-tight">
   392→                <p className="font-semibold text-brand">Auto-pilot</p>
   393→                <p className="text-neutral-400">post ≥95% confidence</p>
   394→              </div>
   395→            </div>
   396→          </div>
   397→
   398→          <div className="mt-4 space-y-4">
   399→            {queue.loading && [0, 1].map((i) => <div key={i} className="h-44 animate-pulse rounded-2xl bg-surface" />)}
   400→            {items.map((item) => (
   401→              <div key={item.id} className={cn("rounded-2xl border p-4", item.urgent ? "border-red-200 bg-red-50/30" : "border-neutral-200 bg-white")}>
   402→                <div className="flex flex-wrap items-center justify-between gap-2">
   403→                  <div className="flex items-center gap-2.5">
   404→                    <StarRating value={item.reviewStars} size={13} />
   405→                    <span className="text-[13px] font-semibold text-brand">{item.author}</span>
   406→                    <span className="text-[11.5px] text-neutral-400">{item.location} · {item.age} old</span>
   407→                    {item.urgent && <span className="inline-flex items-center gap-1 rounded-full bg-red-100 px-2 py-0.5 text-[10.5px] font-bold text-red-600"><AlertTriangle size={10} /> SLA 24h</span>}
   408→                  </div>
   409→                  <div className="flex items-center gap-2">
   410→                    <Badge variant="outline" className="text-[11px] capitalize text-neutral-500">{item.tone}</Badge>
   411→                    <span className="flex items-center gap-1 text-[11.5px] font-semibold text-mint-deep">
   412→                      <span className="h-1.5 w-1.5 rounded-full bg-mint-deep" />
   413→                      {Math.round(item.confidence * 100)}% confident
   414→                    </span>
   415→                  </div>
   416→                </div>
   417→                <p className="mt-2.5 rounded-lg bg-surface px-3 py-2 text-[13px] italic text-neutral-500">“{item.excerpt}”</p>
   418→                <Textarea
   419→                  value={bodyOf(item)}
   420→                  onChange={(e) => setDrafts((d) => ({ ...d, [item.id]: e.target.value }))}
   421→                  className="mt-3 min-h-[72px] border-mint/40 bg-mint-soft/25 text-[13.5px] leading-relaxed focus-visible:ring-mint/50"
   422→                  aria-label={`Suggested reply for ${item.author}`}
   423→                />
   424→                <div className="mt-3 flex flex-wrap items-center gap-2">
   425→                  <Button size="sm" className="h-8 bg-mint-deep text-white hover:bg-mint-deep/90" disabled={busyId === item.id} onClick={() => approve(item)}>
   426→                    {busyId === item.id ? <RefreshCw size={13} className="mr-1 animate-spin" /> : <Check size={13} className="mr-1" />}
   427→                    {busyId === item.id ? "Posting…" : "Approve & post"}
   428→                  </Button>
   429→                  <Button size="sm" variant="outline" className="h-8">
   430→                    <Pencil size={13} className="mr-1" /> Edit
