    56→export const ICON_OPTIONS = ["hotel", "restaurant", "cafe", "clinic", "salon", "gym", "retail", "school"] as const;
    57→
    58→function ListEditor({ spec, value, onChange, canEdit }: {
    59→  spec: Extract<FieldSpec, { type: "list" }>;
    60→  value: Record<string, unknown>[];
    61→  onChange: (next: Record<string, unknown>[]) => void;
    62→  canEdit: boolean;
    63→}) {
    64→  const items = Array.isArray(value) ? value : [];
    65→  const update = (i: number, key: string, v: unknown) =>
    66→    onChange(items.map((it, ix) => (ix === i ? { ...it, [key]: v } : it)));
    67→  const move = (i: number, dir: -1 | 1) => {
    68→    const j = i + dir;
    69→    if (j < 0 || j >= items.length) return;
    70→    const next = [...items];
    71→    [next[i], next[j]] = [next[j], next[i]];
    72→    onChange(next);
    73→  };
    74→  return (
    75→    <div className="space-y-2">
    76→      {items.map((it, i) => (
    77→        <div key={i} className="rounded-xl border border-white/10 bg-brand-deep/50 p-3">
    78→          <div className="mb-2 flex items-center justify-between">
    79→            <span className="text-[10.5px] font-bold uppercase tracking-wider text-white/40">{spec.itemLabel} {i + 1}</span>
    80→            {canEdit && (
    81→              <span className="flex items-center gap-1">
    82→                <button onClick={() => move(i, -1)} className="rounded p-1 text-white/30 hover:bg-white/5 hover:text-white" aria-label="Move up"><ChevronUp size={13} className="rotate-0" /></button>
    83→                <button onClick={() => move(i, 1)} className="rounded p-1 text-white/30 hover:bg-white/5 hover:text-white" aria-label="Move down"><ChevronUp size={13} className="rotate-180" /></button>
    84→                <button onClick={() => onChange(items.filter((_, ix) => ix !== i))} className="rounded p-1 text-white/30 hover:bg-white/5 hover:text-red-300" aria-label="Remove"><X size={13} /></button>
    85→              </span>
    86→            )}
    87→          </div>
    88→          <div className="grid gap-2 sm:grid-cols-2">
    89→            {spec.fields.map((f) => (
    90→              <label key={f.key} className={cn("block", f.type === "area" && "sm:col-span-2")}>
    91→                <span className={lbl}>{f.label}</span>
    92→                {f.type === "area" ? (
    93→                  <textarea value={String(it[f.key] ?? "")} disabled={!canEdit} rows={2}
    94→                    onChange={(e) => update(i, f.key, e.target.value)} className={cn(inpArea, "mt-1")} />
    95→                ) : f.type === "icon" ? (
    96→                  <select value={String(it[f.key] ?? "")} disabled={!canEdit}
    97→                    onChange={(e) => update(i, f.key, e.target.value)} className={cn(inp, "mt-1 appearance-none")}>
    98→                    {ICON_OPTIONS.map((o) => <option key={o} value={o} className="bg-brand">{o}</option>)}
    99→                  </select>
   100→                ) : (
   101→                  <input value={String(it[f.key] ?? "")} disabled={!canEdit}
   102→                    onChange={(e) => update(i, f.key, e.target.value)} className={cn(inp, "mt-1")} />
   103→                )}
   104→              </label>
   105→            ))}
   106→          </div>
   107→        </div>
   108→      ))}
   109→      {canEdit && (
   110→        <Button size="sm" variant="outline" className="h-8 rounded-xl border-white/15 bg-transparent text-[11.5px] text-white/70 hover:bg-white/5 hover:text-white"
   111→          onClick={() => onChange([...items, Object.fromEntries(spec.fields.map((f) => [f.key, ""]))])}>
   112→          <Plus size={12} className="mr-1" /> Add {spec.itemLabel.toLowerCase()}
   113→        </Button>
   114→      )}
   115→    </div>
   116→  );
   117→}

   277→// Normalise array-ish content between editor shapes and renderer shapes.
   278→function toEditorValue(key: string, v: unknown): unknown {
   279→  if (key === "trialCard") {
   280→    const t = (v || {}) as { title?: string; sub?: string; points?: string[] };
   281→    return [{ title: t.title || "", sub: t.sub || "", points: (t.points || []).join("\n") }];
   282→  }
   283→  if (Array.isArray(v)) {
   284→    return v.map((item) => {
   285→      if (item && typeof item === "object") {
   286→        const rec = { ...(item as Record<string, unknown>) };
   287→        for (const k of Object.keys(rec)) {
   288→          if (Array.isArray(rec[k])) rec[k] = (rec[k] as unknown[]).join("\n");
   289→        }
   290→        return rec;
   291→      }
   292→      return { p: String(item) };
   293→    });
   294→  }
   295→  return [{ p: String(v ?? "") }];
   296→}
   297→
   298→function fromEditorValue(key: string, items: Record<string, unknown>[], original: unknown): unknown {
   299→  if (key === "trialCard") {
   300→    const t = (items[0] || {}) as Record<string, string>;
   301→    return { title: t.title || "", sub: t.sub || "", points: (t.points || "").split("\n").map((s) => s.trim()).filter(Boolean) };
   302→  }
   303→  // scalar-string lists (practices, nextSteps, options, topics, story)
   304→  if (Array.isArray(original) && original.length > 0 && typeof original[0] === "string") {
   305→    return items.map((it) => String(it.p ?? it.s ?? it.o ?? it.t ?? "")).filter(Boolean);
   306→  }
   307→  return items.map((it) => {
   308→    const rec: Record<string, unknown> = { ...it };
   309→    for (const k of Object.keys(rec)) {
   310→      if (typeof rec[k] === "string" && (rec[k] as string).includes("\n")) {
   311→        rec[k] = (rec[k] as string).split("\n").map((s) => s.trimEnd()).filter((s, i, a) => !(i === a.length - 1 && s === ""));
   312→      }
   313→    }
   314→    return rec;
   315→  });
   316→}

   776→type CmsMediaRow = { id: string; filename: string; mime: string; size: number; alt: string; url: string; createdAt: string };

   778→function MediaTab({ canEdit, onChanged }: { canEdit: boolean; onChanged: () => void }) {
   779→  const { toast } = useToast();
   780→  const media = useApiData<{ media: CmsMediaRow[] }>("/api/cms/media");
   781→  const [busy, setBusy] = useState(false);
   782→  const rows = media.data?.media || [];
   783→
   784→  const upload = async (file: File) => {
   785→    setBusy(true);
   786→    try {
   787→      const fd = new FormData();
   788→      fd.append("file", file);
   789→      const res = await fetch("/api/cms/media", { method: "POST", body: fd, credentials: "same-origin" });
   790→      const data = await res.json();
   791→      if (!res.ok) throw new Error(data?.error || "Upload failed");
   792→      toast({ title: "Image uploaded", description: data.media.filename });
   793→      media.reload();
   794→      onChanged();
   795→    } catch (e) {
   796→      toast({ title: "Upload failed", description: e instanceof Error ? e.message : "", variant: "destructive" });
   797→    } finally { setBusy(false); }
   798→  };
   799→
   800→  const remove = async (m: CmsMediaRow) => {
   801→    try { await api.del(`/api/cms/media/${m.id}`); media.reload(); onChanged(); toast({ title: "Image deleted", description: m.filename }); }
   802→    catch (e) { toast({ title: "Delete failed", description: e instanceof Error ? e.message : "", variant: "destructive" }); }
   803→  };
   804→
   805→  const copyUrl = async (m: CmsMediaRow) => {
   806→    try { await navigator.clipboard.writeText(m.url); toast({ title: "URL copied", description: m.url }); } catch { toast({ title: m.url }); }
   807→  };
   808→
   809→  return (
   810→    <div className="space-y-3">
   811→      <div className="flex flex-wrap items-center justify-between gap-2">
   812→        <p className="text-[12px] text-white/40">Images usable across the CMS — testimonial photos, blog featured images. Max 2 MB, images only.</p>
   813→        {canEdit && (
   814→          <label className={cn("flex h-8 cursor-pointer items-center rounded-xl bg-mint px-3.5 text-[12px] font-bold text-brand-deep hover:bg-mint/90", busy && "opacity-50")}>
   815→            <ImagePlus size={13} className="mr-1.5" /> {busy ? "Uploading…" : "Upload image"}
   816→            <input type="file" accept="image/*" className="hidden" disabled={busy} onChange={(e) => { const f = e.target.files?.[0]; if (f) upload(f); e.target.value = ""; }} />
   817→          </label>
   818→        )}
   819→      </div>
   820→      {rows.length === 0 && !media.loading ? (
   821→        <div className="rounded-2xl border border-dashed border-white/15 bg-brand-deep/30 p-8 text-center">
   822→          <Palette size={22} className="mx-auto text-white/20" />
   823→          <p className="mt-2 text-[12.5px] text-white/40">No images yet — upload the first one.</p>
   824→        </div>
   825→      ) : (
   826→        <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-4">
   827→          {rows.map((m) => (
   828→            <div key={m.id} className="overflow-hidden rounded-2xl border border-white/10 bg-brand-deep/40">
   829→              <img src={m.url} alt={m.alt || m.filename} className="aspect-video w-full bg-white/5 object-cover" />
   830→              <div className="p-2.5">
   831→                <p className="truncate text-[11.5px] font-semibold text-white">{m.filename}</p>
   832→                <p className="text-[10px] text-white/30">{Math.round(m.size / 1024)} KB · {new Date(m.createdAt).toLocaleDateString("en-IN", { day: "numeric", month: "short" })}</p>
   833→                <div className="mt-1.5 flex gap-1">
   834→                  <button onClick={() => copyUrl(m)} className="rounded-lg bg-white/5 px-2 py-1 text-[10px] font-semibold text-white/60 hover:bg-white/10 hover:text-white">Copy URL</button>

   837→              </div>
   838→            </div>
   839→          ))}
   840→        </div>
   841→      )}
