   725→function TenantDrawer({ tenant, onClose, onChanged, toast, canOperate, isSuper }: {

   728→  onChanged: () => void;

   732→}) {

   737→  const [planPicking, setPlanPicking] = useState(false);
   738→  const detail = useApiData<TenantDetail>(tenant ? `/api/admin/tenants/${tenant.id}` : null);
   739→

   741→    if (!tenant) return;
   742→    setBusy(true);
   743→    try {
   744→      await api.patch(`/api/admin/tenants/${tenant.id}`, data);

   746→      detail.reload();

   751→      setBusy(false);
   752→    }
   753→  };

   756→    if (!tenant) return;
   757→    setBusy(true);

   759→      await api.post(`/api/admin/tenants/${tenant.id}/impersonate`);
   760→      await refresh();

   765→      toast({ title: "Could not start session", description: e instanceof Error ? e.message : "", variant: "destructive" });
   766→    } finally {
   767→      setBusy(false);
   768→    }
   769→  };
   770→
   771→  const t = detail.data?.tenant || tenant;
   772→  const d = detail.data?.detail;
   773→
   774→  return (
   775→    <div className={cn("fixed inset-0 z-50", !open && "pointer-events-none")} aria-hidden={!open}>
