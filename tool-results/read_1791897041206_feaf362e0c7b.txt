   289→type AiAdminPayload = {
   290→  config: { model: string; temperature: number; confidenceThreshold: number; brandVoiceDefault: string; paused: boolean };
   291→  perTenant: { tenant: string; plan: string; drafts: number; classifications: number; tokens: number; costInr: number; quota: number; overQuota: boolean }[];
   292→  quality: { drafts30d: number; published: number; edited: number; pending: number; lowConfidence: number; editRatePct: number };
   293→  totals: { tokens: number; costInr: number; budgetInr: number };

   295→
   296→export function AiAdminView({ canOperate }: { canOperate: boolean }) {
   297→  const { toast } = useToast();
   298→  const data = useApiData<AiAdminPayload>("/api/admin/ai");
   299→  const [draft, setDraft] = useState<{ model: string; temperature: number; confidenceThreshold: number; brandVoiceDefault: string } | null>(null);
   300→  const [busy, setBusy] = useState(false);
   301→
   302→  if (data.loading || !data.data) return <div className="h-64 animate-pulse rounded-2xl bg-brand" />;
   303→  const d = data.data;
   304→  const cfg = d.config;
   305→  const costPct = Math.min(100, Math.round((d.totals.costInr / d.totals.budgetInr) * 100));
   306→
   307→  const save = async (paused?: boolean) => {
   308→    setBusy(true);
   309→    try {
   310→      await api.patch("/api/admin/ai", { ...(draft || cfg), ...(paused !== undefined ? { paused } : {}) });
   311→      toast({ title: "AI engine updated", description: paused !== undefined ? `Generation ${paused ? "paused" : "resumed"} platform-wide.` : "Model settings applied to new drafts." });
   312→      setDraft(null);
   313→      data.reload();
   314→    } catch (e) {
   315→      toast({ title: "Save failed", description: e instanceof Error ? e.message : "", variant: "destructive" });
   316→    } finally {
   317→      setBusy(false);
   318→    }
   319→  };
   320→
   321→  return (
   322→    <div className="space-y-5">
   323→      <div className="grid grid-cols-2 gap-4 xl:grid-cols-4">
   324→        <DarkStat label="Drafts (30d)" value={d.quality.drafts30d.toLocaleString("en-IN")} hint={`${d.quality.published} published · ${d.quality.edited} edited`} icon={<Bot size={15} />} />
   325→        <DarkStat label="Edit rate" value={`${d.quality.editRatePct}%`} hint="share of drafts corrected before publish" icon={<RefreshCw size={15} />} />
   326→        <DarkStat label="Below confidence floor" value={String(d.quality.lowConfidence)} hint={`threshold ${(cfg.confidenceThreshold * 100).toFixed(0)}%`} icon={<AlertTriangle size={15} />} />
   327→        <DarkStat label="Spend (30d)" value={inr(d.totals.costInr)} hint={`${costPct}% of ${inr(d.totals.budgetInr)} budget`} icon={<Bot size={15} />} />
   328→      </div>
   329→
   330→      <Section
   331→        icon={<Bot size={16} className="text-mint" />}
   332→        title="Model settings"
   333→        sub="Applies to every new AI draft and classification across clients"
   334→        right={
   335→          <div className="flex items-center gap-2">
   336→            <span className="text-[11.5px] text-white/50">Generation {cfg.paused ? "paused" : "live"}</span>
   337→            <Switch checked={!cfg.paused} disabled={!canOperate || busy} onCheckedChange={(v) => save(!v)} />
   338→          </div>
   339→        }
   340→      >
   341→        <div className="grid gap-3 lg:grid-cols-4">
   342→          <label className="block">
   343→            <span className="text-[10.5px] font-semibold uppercase tracking-wider text-white/40">Model</span>
   344→            <select value={draft?.model ?? cfg.model} onChange={(e) => setDraft({ ...(draft || cfg), model: e.target.value })} className={cn(inputCls, "mt-1")}>
   345→              <option value="glm-4-flash">GLM-4-Flash (fast)</option>
   346→              <option value="glm-4-plus">GLM-4-Plus (quality)</option>
   347→              <option value="fallback-template">Template fallback (no LLM)</option>
   348→            </select>
   349→          </label>
