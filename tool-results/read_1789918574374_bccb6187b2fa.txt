    72→
    73→  const overview = useApiData<Overview>(view === "overview" ? "/api/overview" : null);
    74→  const badgeCounts = useApiData<Overview>("/api/overview");

    79→    if (user && !user.tenant?.onboardingDone) setSetupOpen(true);
    80→  }, [user?.tenant?.onboardingDone]);

    83→    navigate({ surface: "client", view: v });

    85→  };
    86→

    88→  const [approving, setApproving] = useState<string | null>(null);

    90→    setApproving(r.id);
    91→    try {
    92→      await api.post(`/api/reviews/${r.id}/reply`, { body });
    93→      toast({ title: "Reply posted", description: `Published to Google for ${r.author}.` });

    98→      toast({ title: "Could not post reply", description: e instanceof Error ? e.message : "", variant: "destructive" });
    99→    } finally {
   100→      setApproving(null);
   101→    }
   102→  };
   103→  const generateDraft = async (r: ReviewVM) => {
   104→    setApproving(r.id);
   105→    try {
   106→      await api.post(`/api/reviews/${r.id}/draft`);
   107→      toast({ title: "Draft ready", description: `AI reply drafted for ${r.author}.` });
   108→      overview.reload();
   109→      badgeCounts.reload();

   111→      toast({ title: "Draft failed", description: e instanceof Error ? e.message : "", variant: "destructive" });

   113→      setApproving(null);
   114→    }
   115→  };

   125→          <Shieldish /> Viewing {user?.tenant?.name || "tenant"} workspace as support — actions are audit-logged

   127→            className="rounded-lg bg-brand-deep/10 px-2.5 py-1 font-bold hover:bg-brand-deep/20"

   132→            }}

   134→            Exit session <Xish />
   135→          </button>
   136→        </div>
   137→      )}

   139→      {/* Sidebar */}
   140→      <aside
   141→        className={cn(
   142→          "fixed inset-y-0 left-0 z-40 flex w-[240px] flex-col border-r border-neutral-200/60 bg-white transition-transform lg:static lg:translate-x-0",
   143→          user?.impersonatedBy ? "pt-9" : "",
   144→          navOpen ? "translate-x-0" : "-translate-x-full"
   145→        )}
   146→      >
   147→        <div className="px-5 pb-4 pt-5">
   148→          <BrandLogo height={32} />
   149→        </div>
   150→        <div className="mx-4 mb-4 rounded-xl bg-surface p-3">
   151→          <p className="text-[10.5px] font-semibold uppercase tracking-wider text-neutral-400">Workspace</p>
   152→          <div className="mt-1 flex items-center justify-between text-left">
   153→            <span className="truncate text-[13px] font-semibold">{user?.tenant?.name || "Workspace"}</span>
   154→            <ChevronDown size={14} className="shrink-0 text-neutral-400" />
   155→          </div>
   156→        </div>
   157→        <nav className="flex-1 space-y-1 px-3" aria-label="Dashboard sections">

   162→              className={cn(

   165→              )}

   173→                  view === n.key ? "bg-white/20 text-white" : "bg-amber-soft text-amber-deep"
   174→                )}>
   175→                  {badgeCounts.data?.kpis.drafts}
   176→                </span>
   177→              )}
   178→              {n.key === "reviews" && (badgeCounts.data?.kpis.alerts ?? 0) > 0 && (
   179→                <span className={cn(
   180→                  "ml-auto rounded-full px-1.5 py-0.5 text-[10.5px] font-bold",

   259→              <OverviewView
   260→                data={overview.data}
   261→                loading={overview.loading}

   264→                onOpenReview={setSelectedReview}
   265→                onApprove={approveReply}

   287→        busy={approving === selectedReview?.id}
   288→      />
   289→      <LocationDrawer location={selectedLoc} onClose={() => { setSelectedLoc(null); badgeCounts.reload(); }} />
   290→    </div>
   291→  );

   443→function ReviewsView({ onOpenReview, onApprove, onDraft, busy, locations }: {
   444→  onOpenReview: (r: ReviewVM) => void;
   445→  onApprove: (r: ReviewVM, body: string) => void;
   446→  onDraft: (r: ReviewVM) => void;
   447→  busy: string | null;
   448→  locations: LocationVM[];
   449→}) {
   450→  const { toast } = useToast();
   451→  const [status, setStatus] = useState<StatusFilter>("all");
   452→  const [minStars, setMinStars] = useState(0);
   453→  const [syncing, setSyncing] = useState(false);
   454→  const reviews = useApiData<{ reviews: ReviewVM[]; total: number }>(
   455→    `/api/reviews?status=${status}&minStars=${minStars}`
   456→  );
   457→  const filtered = useMemo(() => reviews.data?.reviews || [], [reviews.data]);
   458→
   459→  const syncAll = async () => {
   460→    setSyncing(true);
   461→    try {
   462→      let created = 0;
   463→      for (const loc of locations) {
   464→        const res = await api.post<{ newReviews: number }>(`/api/locations/${loc.id}/sync`);
   465→        created += res.newReviews;
   466→      }
   467→      toast({ title: "Sync complete", description: created ? `${created} new review${created > 1 ? "s" : ""} pulled and scored.` : "All profiles up to date — no new reviews." });
   468→      reviews.reload();
   469→    } catch (e) {
   470→      toast({ title: "Sync failed", description: e instanceof Error ? e.message : "", variant: "destructive" });
   471→    } finally {
   472→      setSyncing(false);
   473→    }
   474→  };
   475→
   476→  const exportCsv = () => {
   477→    const rows = [
   478→      ["Author", "Stars", "Location", "Sentiment", "Status", "Text", "Received"],
   479→      ...filtered.map((r) => [r.author, String(r.stars), r.location, r.sentiment, r.status, `"${r.text.replace(/"/g, '""')}"`, r.createdAt]),
   480→    ];
   481→    const csv = rows.map((r) => r.join(",")).join("\n");
   482→    const blob = new Blob([csv], { type: "text/csv" });
   483→    const url = URL.createObjectURL(blob);
   484→    const a = document.createElement("a");
   485→    a.href = url;
   486→    a.download = `sensiblee-reviews-${new Date().toISOString().slice(0, 10)}.csv`;
   487→    a.click();
   488→    URL.revokeObjectURL(url);
   489→    toast({ title: "Export downloaded", description: `${filtered.length} reviews in the current filter.` });
   490→  };
   491→
   492→  return (
   493→    <div className="space-y-5">
   494→      <div className="flex flex-wrap items-end justify-between gap-3">
   495→        <div>
   496→          <h1 className="text-[22px] font-bold tracking-tight">Review monitor</h1>
   497→          <p className="mt-0.5 text-[13.5px] text-neutral-500">Module A · every Google review across your locations, synced every 15 minutes.</p>
   498→        </div>
   499→        <div className="flex gap-2">
   500→          <Button variant="outline" size="sm" className="h-9 rounded-xl text-[12.5px]" onClick={syncAll} disabled={syncing}>
   501→            <RefreshCw size={13} className={cn("mr-1.5", syncing && "animate-spin")} />
   502→            {syncing ? "Syncing…" : "Sync all profiles"}
   503→          </Button>
   504→          <Button variant="outline" size="sm" className="h-9 rounded-xl text-[12.5px]" onClick={exportCsv}>
   505→            <Download size={13} className="mr-1.5" /> Export CSV
   506→          </Button>
   507→        </div>
   508→      </div>
   509→
   510→      <div className="flex flex-wrap items-center gap-2 rounded-2xl border border-neutral-200/70 bg-white p-3">
   511→        <div className="flex flex-wrap gap-1.5">
   512→          {(["all", "ai-draft", "replied", "awaiting"] as StatusFilter[]).map((s) => (
   513→            <button
   514→              key={s}
   515→              onClick={() => setStatus(s)}
   516→              className={cn(
   517→                "rounded-full px-3 py-1.5 text-[12.5px] font-medium transition-colors",
   518→                status === s ? "bg-brand text-white" : "bg-surface text-neutral-500 hover:text-brand"
   519→              )}
   520→            >
   521→              {s === "all" ? "All" : s === "ai-draft" ? "AI drafts" : s === "replied" ? "Replied" : "Awaiting"}
   522→            </button>
   523→          ))}
   524→        </div>
   525→        <div className="ml-auto flex items-center gap-2">
   526→          <span className="text-[12px] text-neutral-400">Min stars</span>
   527→          {[0, 1, 2, 3, 4].map((n) => (
   528→            <button
   529→              key={n}
   530→              onClick={() => setMinStars(n)}
   531→              className={cn("h-7 min-w-7 rounded-lg px-1.5 text-[12px] font-semibold", minStars === n ? "bg-mint-soft text-mint-deep ring-1 ring-mint/50" : "text-neutral-400 hover:text-brand")}
   532→            >
   533→              {n === 0 ? "Any" : `${n}+`}
   534→            </button>
   535→          ))}
   536→        </div>
   537→      </div>
   538→
   539→      <div className="space-y-3">
   540→        {reviews.loading && <div className="space-y-3">{[0, 1, 2].map((i) => <div key={i} className="h-28 animate-pulse rounded-2xl bg-surface" />)}</div>}
   541→        {!reviews.loading && filtered.map((r) => (
   542→          <ReviewRow key={r.id} review={r} onOpen={onOpenReview} onApprove={onApprove} onDraft={onDraft} approving={busy} />
   543→        ))}
   544→        {!reviews.loading && filtered.length === 0 && (
   545→          <div className="rounded-2xl border border-dashed border-neutral-200 p-10 text-center text-[13.5px] text-neutral-400">
   546→            No reviews match this filter.
   547→          </div>
   548→        )}
   549→      </div>
   550→    </div>
   551→  );
   552→}

   554→function ReviewRow({ review: r, compact = false, onOpen, onApprove, onDraft, approving }: {
   555→  review: ReviewVM;
   556→  compact?: boolean;
   557→  onOpen?: (r: ReviewVM) => void;
   558→  onApprove?: (r: ReviewVM, body: string) => void;
   559→  onDraft?: (r: ReviewVM) => void;
   560→  approving?: string | null;
   561→}) {
   562→  const [expanded, setExpanded] = useState(false);
   563→  const isBusy = approving === r.id;
   564→  return (
   565→    <div
   566→      onClick={() => onOpen?.(r)}
   567→      role={onOpen ? "button" : undefined}
   568→      tabIndex={onOpen ? 0 : undefined}
   569→      onKeyDown={onOpen ? (e) => { if (e.key === "Enter" || e.key === " ") { e.preventDefault(); onOpen(r); } } : undefined}
   570→      className={cn(
   571→        "rounded-2xl border bg-white p-4 shadow-[0_1px_2px_rgba(16,24,40,0.04)] transition-shadow hover:shadow-md",
   572→        r.stars <= 2 ? "border-red-200/60" : "border-neutral-200/70",
   573→        onOpen && "cursor-pointer focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-mint"
   574→      )}
   575→    >
   576→      <div className="flex items-start gap-3">
   577→        <span className={cn(
   578→          "flex h-9 w-9 shrink-0 items-center justify-center rounded-full text-[11.5px] font-bold text-white",
   579→          r.sentiment === "positive" ? "bg-mint-deep" : r.sentiment === "neutral" ? "bg-amber-500" : "bg-red-500"
   580→        )}>
   581→          {r.initials}
   582→        </span>
   583→        <div className="min-w-0 flex-1">
   584→          <div className="flex flex-wrap items-center gap-x-2 gap-y-1">
   585→            <span className="text-[13.5px] font-semibold text-brand">{r.author}</span>
   586→            <StarRating value={r.stars} size={12} />
   587→            <span className="text-[11.5px] text-neutral-400">{r.location} · {r.time}</span>
   588→            <span className="ml-auto flex items-center gap-1.5">
   589→              <GoogleG size={13} />
   590→              <SentimentBadge sentiment={r.sentiment} />
   591→              <StatusBadge status={r.status} />
   592→            </span>
   593→          </div>
   594→          <p className={cn("mt-1.5 text-[13.5px] leading-relaxed text-neutral-600", compact && !expanded && "line-clamp-2")}>{r.text}</p>
   595→          <div className="mt-2 flex flex-wrap items-center gap-1.5">
   596→            {r.keywords.map((kw) => (
   597→              <span key={kw} className="rounded-full bg-surface px-2 py-0.5 text-[11px] font-medium text-neutral-500">{kw}</span>
   598→            ))}
   599→          </div>
   600→
   601→          {(r.draft || r.reply) && (
   602→            <div className={cn("mt-3 rounded-xl p-3", r.status === "ai-draft" ? "bg-mint-soft/40 ring-1 ring-mint/30" : "bg-surface")}>
   603→              <p className="flex items-center gap-1.5 text-[11px] font-semibold uppercase tracking-wide text-neutral-400">
   604→                {r.status === "ai-draft" ? <><Sparkles size={11} className="text-mint-deep" /> AI suggested reply · {Math.round((r.confidence ?? 0.9) * 100)}% confident</> : "Your reply"}
   605→              </p>
   606→              <p className={cn("mt-1 text-[13px] leading-relaxed text-neutral-600", compact && !expanded && "line-clamp-2")}>{r.draft ?? r.reply}</p>
   607→            </div>
   608→          )}
   609→
   610→          <div className="mt-2.5 flex items-center gap-2">
   611→            {(r.draft || compact) && (
   612→              <button
   613→                className="text-[12px] font-semibold text-mint-deep hover:underline"
   614→                onClick={(e) => { e.stopPropagation(); setExpanded((x) => !x); }}
   615→              >
   616→                {expanded ? "Show less" : r.draft ? "Review draft" : "Show more"}
   617→              </button>
   618→            )}
   619→            {r.status === "ai-draft" && onApprove && (
   620→              <Button size="sm" className="h-7 rounded-lg bg-mint-deep px-2.5 text-[11.5px] text-white hover:bg-mint-deep/90" disabled={isBusy}
   621→                onClick={(e) => { e.stopPropagation(); onApprove(r, r.draft || ""); }}>
   622→                {isBusy ? <RefreshCw size={12} className="mr-1 animate-spin" /> : <Check size={12} className="mr-1" />}
   623→                {isBusy ? "Posting…" : "Approve"}
   624→              </Button>
   625→            )}
   626→            {r.status === "awaiting" && onDraft && (
   627→              <Button size="sm" variant="outline" className="h-7 rounded-lg px-2.5 text-[11.5px]" disabled={isBusy}
   628→                onClick={(e) => { e.stopPropagation(); onDraft(r); }}>
   629→                <Sparkles size={12} className="mr-1" /> {isBusy ? "Writing…" : "Draft a reply"}
   630→              </Button>
   631→            )}
   632→            {onOpen && (
   633→              <button
   634→                className="text-[12px] font-semibold text-neutral-400 hover:text-brand"
   635→                onClick={(e) => { e.stopPropagation(); onOpen(r); }}
   636→              >
   637→                Details
   638→              </button>
   639→            )}
   640→            <span className="ml-auto text-[11px] text-neutral-300">via {r.source === "request" ? "request · Google" : "Google"}</span>
   641→          </div>
   642→        </div>
   643→      </div>
   644→    </div>
   645→  );
   646→}
