   276→function QrKitCard({ locations, onPreviewGuest }: { locations: CampaignsPayload["locations"]; onPreviewGuest: () => void }) {
   277→  const { toast } = useToast();
   278→  const firstCode = locations[0]?.code;
   279→  const qr = useApiData<{ dataUrl: string; url: string }>(firstCode ? `/api/qr?code=${firstCode}` : null);
   280→
   281→  const copy = async () => {
   282→    if (!qr.data) return;
   283→    try {
   284→      await navigator.clipboard.writeText(qr.data.url);
   285→      toast({ title: "Link copied", description: qr.data.url });
   286→    } catch {
   287→      toast({ title: "Copy failed", description: qr.data.url });
   288→    }
   289→  };
   290→
   291→  const download = () => {
   292→    if (!qr.data) return;
   293→    const a = document.createElement("a");
   294→    a.href = qr.data.dataUrl;
   295→    a.download = `sensiblee-qr-${firstCode}.png`;
   296→    a.click();
   297→    toast({ title: "QR kit downloading", description: "Print-ready PNG · 480 px" });
   298→  };
   299→
   300→  return (
   301→    <ChartCard title="Table QR kit" subtitle="Review Growth Engine · live scannable code">
   302→      <div className="flex items-center gap-4">
   303→        <div className="rounded-2xl border border-neutral-200 bg-cream p-2.5">
   304→          {qr.data ? (
   305→            <img src={qr.data.dataUrl} alt={`QR code for ${locations[0]?.name || "location"}`} width={124} height={124} className="rounded-lg" />
   306→          ) : (
   307→            <div className="h-[124px] w-[124px] animate-pulse rounded-lg bg-surface" />
   308→          )}
   309→        </div>
   310→        <div className="flex-1 text-[12.5px] leading-relaxed text-neutral-500">
   311→          <p className="font-semibold text-brand">scan → rate us in 2 taps</p>
   312→          <p className="mt-1">
   313→            Branded landing with your logo, star pre-select and Google hand-off. Printed at 40×40 mm on table tents.
   314→          </p>
   315→          <p className="mt-1 truncate font-mono text-[10.5px] text-neutral-400">{qr.data?.url || "…"}</p>
   316→          <div className="mt-3 flex flex-wrap gap-2">
   317→            <Button size="sm" variant="outline" className="h-8 text-xs" onClick={download} disabled={!qr.data}>

   335→type QueueItem = {
   336→  id: string; reviewStars: number; author: string; location: string; excerpt: string;
   337→  draft: string; tone: string; confidence: number; age: string; urgent?: boolean; text: string;
   338→};
   339→
   340→export function AiAssistantView() {
   341→  const { toast } = useToast();
   342→  const queue = useApiData<{ queue: QueueItem[] }>("/api/ai/queue");
   343→  const analytics = useApiData<{ kpis: { draftsPending: number; draftsApproved30d: number; atRisk: number; medianResponse: string } }>("/api/analytics");
   344→  const [drafts, setDrafts] = useState<Record<string, string>>({});
   345→  const [autoPilot, setAutoPilot] = useState(false);
   346→  const [busyId, setBusyId] = useState<string | null>(null);
   347→  const [approvedIds, setApprovedIds] = useState<string[]>([]);
   348→
   349→  const items = (queue.data?.queue || []).filter((i) => !approvedIds.includes(i.id));
   350→  const k = analytics.data?.kpis;
   351→
   352→  const bodyOf = (item: QueueItem) => drafts[item.id] ?? item.draft;
   353→
   354→  const approve = async (item: QueueItem) => {
   355→    setBusyId(item.id);
   356→    try {
   357→      await api.post(`/api/reviews/${item.id}/reply`, { body: bodyOf(item) });
   358→      setApprovedIds((a) => [...a, item.id]);
   359→      toast({ title: "Reply posted", description: `Published to Google for ${item.author}'s review.` });
   360→      queue.reload();
   361→    } catch (e) {
   362→      toast({ title: "Could not post", description: e instanceof Error ? e.message : "", variant: "destructive" });
   363→    } finally {
   364→      setBusyId(null);
   365→    }
   366→  };
   367→
   368→  const regenerate = async (item: QueueItem) => {
   369→    setBusyId(item.id);
   370→    try {
   371→      toast({ title: "Regenerating draft", description: "The AI worker is rewriting this reply in your voice." });
   372→      const res = await api.post<{ review: ReviewVM }>(`/api/reviews/${item.id}/draft`);
   373→      if (res.review.draft) setDrafts((d) => ({ ...d, [item.id]: res.review.draft! }));
   374→      queue.reload();
   375→    } catch (e) {
   376→      toast({ title: "Regeneration failed", description: e instanceof Error ? e.message : "", variant: "destructive" });
   377→    } finally {
   378→      setBusyId(null);
   379→    }
   380→  };
   381→
   382→  const toggleAutoPilot = async (v: boolean) => {
   383→    setAutoPilot(v);
   384→    try {
   385→      await api.patch("/api/settings", { ai: { threshold: v ? 95 : 90 } });
   386→      toast({
   387→        title: v ? "Auto-pilot staged" : "Auto-pilot off",
   388→        description: v
   389→          ? "Replies above 95% confidence would post automatically once you enable posting — 1–2★ reviews always wait for you."
   390→          : "Back to manual approval for all replies.",
   391→      });
   392→    } catch {
   393→      setAutoPilot(!v);
   394→      toast({ title: "Could not update", description: "Try again in a moment.", variant: "destructive" });
   395→    }
   396→  };
   397→
   398→  return (
   399→    <div className="space-y-5">
   400→      <div className="grid gap-5 lg:grid-cols-3">
   401→        <div className="lg:col-span-2 rounded-2xl border border-neutral-200/70 bg-white p-5">
   402→          <div className="flex flex-wrap items-center justify-between gap-3">
   403→            <div>
   404→              <h3 className="flex items-center gap-2 text-[15px] font-semibold text-brand">
   405→                <Sparkles size={16} className="text-mint-deep" /> Approval queue
   406→                <Badge className="bg-amber-soft text-amber-deep hover:bg-amber-soft">{items.length} pending</Badge>
   407→              </h3>
   408→              <p className="mt-0.5 text-xs text-neutral-400">AI Reputation Copilot · every reply is yours to approve, edit or regenerate</p>
   409→            </div>
   410→            <div className="flex items-center gap-2.5 rounded-xl bg-surface px-3 py-2">
   411→              <Switch checked={autoPilot} onCheckedChange={toggleAutoPilot} />
   412→              <div className="text-[12px] leading-tight">
   413→                <p className="font-semibold text-brand">Auto-pilot</p>
   414→                <p className="text-neutral-400">post ≥95% confidence</p>
   415→              </div>
   416→            </div>
   417→          </div>
   418→
   419→          <div className="mt-4 space-y-4">
   420→            {queue.loading && [0, 1].map((i) => <div key={i} className="h-44 animate-pulse rounded-2xl bg-surface" />)}
   421→            {items.map((item) => (
   422→              <div key={item.id} className={cn("rounded-2xl border p-4", item.urgent ? "border-red-200 bg-red-50/30" : "border-neutral-200 bg-white")}>
   423→                <div className="flex flex-wrap items-center justify-between gap-2">
   424→                  <div className="flex items-center gap-2.5">
   425→                    <StarRating value={item.reviewStars} size={13} />
   426→                    <span className="text-[13px] font-semibold text-brand">{item.author}</span>
   427→                    <span className="text-[11.5px] text-neutral-400">{item.location} · {item.age} old</span>
   428→                    {item.urgent && <span className="inline-flex items-center gap-1 rounded-full bg-red-100 px-2 py-0.5 text-[10.5px] font-bold text-red-600"><AlertTriangle size={10} /> SLA 24h</span>}
   429→                  </div>
   430→                  <div className="flex items-center gap-2">
   431→                    <Badge variant="outline" className="text-[11px] capitalize text-neutral-500">{item.tone}</Badge>
   432→                    <span className="flex items-center gap-1 text-[11.5px] font-semibold text-mint-deep">
   433→                      <span className="h-1.5 w-1.5 rounded-full bg-mint-deep" />
