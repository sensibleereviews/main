    52→  | { key: string; label: string; type: "text" | "area"; hint?: string }

    55→export const ICON_OPTIONS = ["hotel", "restaurant", "cafe", "clinic", "salon", "gym", "retail", "school"] as const;
    56→
    57→function ListEditor({ spec, value, onChange, canEdit }: {
    58→  spec: Extract<FieldSpec, { type: "list" }>;
    59→  value: Record<string, unknown>[];
    60→  onChange: (next: Record<string, unknown>[]) => void;
    61→  canEdit: boolean;
    62→}) {
    63→  const items = Array.isArray(value) ? value : [];
    64→  const update = (i: number, key: string, v: unknown) =>
    65→    onChange(items.map((it, ix) => (ix === i ? { ...it, [key]: v } : it)));
    66→  const move = (i: number, dir: -1 | 1) => {
    67→    const j = i + dir;
    68→    if (j < 0 || j >= items.length) return;
    69→    const next = [...items];
    70→    [next[i], next[j]] = [next[j], next[i]];
    71→    onChange(next);
    72→  };
    73→  return (
    74→    <div className="space-y-2">
    75→      {items.map((it, i) => (
    76→        <div key={i} className="rounded-xl border border-white/10 bg-brand-deep/50 p-3">
    77→          <div className="mb-2 flex items-center justify-between">
    78→            <span className="text-[10.5px] font-bold uppercase tracking-wider text-white/40">{spec.itemLabel} {i + 1}</span>
    79→            {canEdit && (
    80→              <span className="flex items-center gap-1">
    81→                <button onClick={() => move(i, -1)} className="rounded p-1 text-white/30 hover:bg-white/5 hover:text-white" aria-label="Move up"><ChevronUp size={13} className="rotate-0" /></button>
    82→                <button onClick={() => move(i, 1)} className="rounded p-1 text-white/30 hover:bg-white/5 hover:text-white" aria-label="Move down"><ChevronUp size={13} className="rotate-180" /></button>
    83→                <button onClick={() => onChange(items.filter((_, ix) => ix !== i))} className="rounded p-1 text-white/30 hover:bg-white/5 hover:text-red-300" aria-label="Remove"><X size={13} /></button>
    84→              </span>
    85→            )}
    86→          </div>
    87→          <div className="grid gap-2 sm:grid-cols-2">
    88→            {spec.fields.map((f) => (
    89→              <label key={f.key} className={cn("block", f.type === "area" && "sm:col-span-2")}>
    90→                <span className={lbl}>{f.label}</span>
    91→                {f.type === "area" ? (
    92→                  <textarea value={String(it[f.key] ?? "")} disabled={!canEdit} rows={2}
    93→                    onChange={(e) => update(i, f.key, e.target.value)} className={cn(inpArea, "mt-1")} />
    94→                ) : f.type === "icon" ? (
    95→                  <select value={String(it[f.key] ?? "")} disabled={!canEdit}
    96→                    onChange={(e) => update(i, f.key, e.target.value)} className={cn(inp, "mt-1 appearance-none")}>
    97→                    {ICON_OPTIONS.map((o) => <option key={o} value={o} className="bg-brand">{o}</option>)}
    98→                  </select>
    99→                ) : (
   100→                  <input value={String(it[f.key] ?? "")} disabled={!canEdit}
   101→                    onChange={(e) => update(i, f.key, e.target.value)} className={cn(inp, "mt-1")} />
   102→                )}
   103→              </label>
   104→            ))}
   105→          </div>
   106→        </div>
   107→      ))}
   108→      {canEdit && (
   109→        <Button size="sm" variant="outline" className="h-8 rounded-xl border-white/15 bg-transparent text-[11.5px] text-white/70 hover:bg-white/5 hover:text-white"
   110→          onClick={() => onChange([...items, Object.fromEntries(spec.fields.map((f) => [f.key, ""]))])}>
   111→          <Plus size={12} className="mr-1" /> Add {spec.itemLabel.toLowerCase()}
   112→        </Button>
   113→      )}
   114→    </div>
   115→  );
   116→}

   272→// Normalise array-ish content between editor shapes and renderer shapes.
   273→function toEditorValue(key: string, v: unknown): unknown {
   274→  if (key === "trialCard") {
   275→    const t = (v || {}) as { title?: string; sub?: string; points?: string[] };
   276→    return [{ title: t.title || "", sub: t.sub || "", points: (t.points || []).join("\n") }];
   277→  }
   278→  if (Array.isArray(v)) {
   279→    return v.map((item) => {
   280→      if (item && typeof item === "object") {
   281→        const rec = { ...(item as Record<string, unknown>) };
   282→        for (const k of Object.keys(rec)) {
   283→          if (Array.isArray(rec[k])) rec[k] = (rec[k] as unknown[]).join("\n");
   284→        }
   285→        return rec;
   286→      }
   287→      return { p: String(item) };
   288→    });
   289→  }
   290→  return [{ p: String(v ?? "") }];
   291→}
   292→
   293→function fromEditorValue(key: string, items: Record<string, unknown>[], original: unknown): unknown {
   294→  if (key === "trialCard") {
   295→    const t = (items[0] || {}) as Record<string, string>;
   296→    return { title: t.title || "", sub: t.sub || "", points: (t.points || "").split("\n").map((s) => s.trim()).filter(Boolean) };
   297→  }
   298→  // scalar-string lists (practices, nextSteps, options, topics, story)
   299→  if (Array.isArray(original) && original.length > 0 && typeof original[0] === "string") {
   300→    return items.map((it) => String(it.p ?? it.s ?? it.o ?? it.t ?? "")).filter(Boolean);
   301→  }
   302→  return items.map((it) => {
   303→    const rec: Record<string, unknown> = { ...it };
   304→    for (const k of Object.keys(rec)) {
   305→      if (typeof rec[k] === "string" && (rec[k] as string).includes("\n")) {
   306→        rec[k] = (rec[k] as string).split("\n").map((s) => s.trimEnd()).filter((s, i, a) => !(i === a.length - 1 && s === ""));
   307→      }
   308→    }
   309→    return rec;
   310→  });
   311→}

   745→function MediaTab({ canEdit, onChanged }: { canEdit: boolean; onChanged: () => void }) {
   746→  const { toast } = useToast();
   747→  const media = useApiData<{ media: CmsMediaRow[] }>("/api/cms/media");
   748→  const [busy, setBusy] = useState(false);
   749→  const rows = media.data?.media || [];
   750→
   751→  const upload = async (file: File) => {
   752→    setBusy(true);
   753→    try {
   754→      const fd = new FormData();
   755→      fd.append("file", file);
   756→      const res = await fetch("/api/cms/media", { method: "POST", body: fd, credentials: "same-origin" });
   757→      const data = await res.json();
   758→      if (!res.ok) throw new Error(data?.error || "Upload failed");
   759→      toast({ title: "Image uploaded", description: data.media.filename });
   760→      media.reload();
   761→      onChanged();
   762→    } catch (e) {
   763→      toast({ title: "Upload failed", description: e instanceof Error ? e.message : "", variant: "destructive" });
   764→    } finally { setBusy(false); }
   765→  };
   766→
   767→  const remove = async (m: CmsMediaRow) => {
   768→    try { await api.del(`/api/cms/media/${m.id}`); media.reload(); onChanged(); toast({ title: "Image deleted", description: m.filename }); }
   769→    catch (e) { toast({ title: "Delete failed", description: e instanceof Error ? e.message : "", variant: "destructive" }); }
   770→  };
   771→
   772→  const copyUrl = async (m: CmsMediaRow) => {
   773→    try { await navigator.clipboard.writeText(m.url); toast({ title: "URL copied", description: m.url }); } catch { toast({ title: m.url }); }
   774→  };

   776→  return (
   777→    <div className="space-y-3">
   778→      <div className="flex flex-wrap items-center justify-between gap-2">
   779→        <p className="text-[12px] text-white/40">Images usable across the CMS — testimonial photos, blog featured images. Max 2 MB, images only.</p>

   782→            <ImagePlus size={13} className="mr-1.5" /> {busy ? "Uploading…" : "Upload image"}
   783→            <input type="file" accept="image/*" className="hidden" disabled={busy} onChange={(e) => { const f = e.target.files?.[0]; if (f) upload(f); e.target.value = ""; }} />
   784→          </label>
   785→        )}
   786→      </div>
   787→      {rows.length === 0 && !media.loading ? (
   788→        <div className="rounded-2xl border border-dashed border-white/15 bg-brand-deep/30 p-8 text-center">
   789→          <Palette size={22} className="mx-auto text-white/20" />
