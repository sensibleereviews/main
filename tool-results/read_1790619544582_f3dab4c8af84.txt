    57→function ListEditor({ spec, value, onChange, canEdit }: {
    58→  spec: Extract<FieldSpec, { type: "list" }>;
    59→  value: Record<string, unknown>[];
    60→  onChange: (next: Record<string, unknown>[]) => void;
    61→  canEdit: boolean;
    62→}) {
    63→  const items = Array.isArray(value) ? value : [];
    64→  const update = (i: number, key: string, v: unknown) =>
    65→    onChange(items.map((it, ix) => (ix === i ? { ...it, [key]: v } : it)));
    66→  const move = (i: number, dir: -1 | 1) => {
    67→    const j = i + dir;
    68→    if (j < 0 || j >= items.length) return;
    69→    const next = [...items];
    70→    [next[i], next[j]] = [next[j], next[i]];
    71→    onChange(next);
    72→  };
    73→  return (
    74→    <div className="space-y-2">
    75→      {items.map((it, i) => (
    76→        <div key={i} className="rounded-xl border border-white/10 bg-brand-deep/50 p-3">
    77→          <div className="mb-2 flex items-center justify-between">
    78→            <span className="text-[10.5px] font-bold uppercase tracking-wider text-white/40">{spec.itemLabel} {i + 1}</span>
    79→            {canEdit && (
    80→              <span className="flex items-center gap-1">
    81→                <button onClick={() => move(i, -1)} className="rounded p-1 text-white/30 hover:bg-white/5 hover:text-white" aria-label="Move up"><ChevronUp size={13} className="rotate-0" /></button>
    82→                <button onClick={() => move(i, 1)} className="rounded p-1 text-white/30 hover:bg-white/5 hover:text-white" aria-label="Move down"><ChevronUp size={13} className="rotate-180" /></button>
    83→                <button onClick={() => onChange(items.filter((_, ix) => ix !== i))} className="rounded p-1 text-white/30 hover:bg-white/5 hover:text-red-300" aria-label="Remove"><X size={13} /></button>
    84→              </span>
    85→            )}
    86→          </div>
    87→          <div className="grid gap-2 sm:grid-cols-2">
    88→            {spec.fields.map((f) => (
    89→              <label key={f.key} className={cn("block", f.type === "area" && "sm:col-span-2")}>
    90→                <span className={lbl}>{f.label}</span>
    91→                {f.type === "area" ? (
    92→                  <textarea value={String(it[f.key] ?? "")} disabled={!canEdit} rows={2}
    93→                    onChange={(e) => update(i, f.key, e.target.value)} className={cn(inpArea, "mt-1")} />
    94→                ) : f.type === "icon" ? (
    95→                  <select value={String(it[f.key] ?? "")} disabled={!canEdit}
    96→                    onChange={(e) => update(i, f.key, e.target.value)} className={cn(inp, "mt-1 appearance-none")}>
    97→                    {ICON_OPTIONS.map((o) => <option key={o} value={o} className="bg-brand">{o}</option>)}
    98→                  </select>
    99→                ) : (
   100→                  <input value={String(it[f.key] ?? "")} disabled={!canEdit}
   101→                    onChange={(e) => update(i, f.key, e.target.value)} className={cn(inp, "mt-1")} />
   102→                )}
   103→              </label>
   104→            ))}
   105→          </div>
   106→        </div>
   107→      ))}
   108→      {canEdit && (
   109→        <Button size="sm" variant="outline" className="h-8 rounded-xl border-white/15 bg-transparent text-[11.5px] text-white/70 hover:bg-white/5 hover:text-white"
   110→          onClick={() => onChange([...items, Object.fromEntries(spec.fields.map((f) => [f.key, ""]))])}>
   111→          <Plus size={12} className="mr-1" /> Add {spec.itemLabel.toLowerCase()}
   112→        </Button>
   113→      )}
   114→    </div>
   115→  );
   116→}

   260→// Normalise array-ish content between editor shapes and renderer shapes.
   261→function toEditorValue(key: string, v: unknown): unknown {
   262→  if (key === "trialCard") {
   263→    const t = (v || {}) as { title?: string; sub?: string; points?: string[] };
   264→    return [{ title: t.title || "", sub: t.sub || "", points: (t.points || []).join("\n") }];
   265→  }
   266→  if (Array.isArray(v)) {
   267→    return v.map((item) => {
   268→      if (item && typeof item === "object") {
   269→        const rec = { ...(item as Record<string, unknown>) };
   270→        for (const k of Object.keys(rec)) {
   271→          if (Array.isArray(rec[k])) rec[k] = (rec[k] as unknown[]).join("\n");
   272→        }
   273→        return rec;
   274→      }
   275→      return { p: String(item) };
   276→    });
   277→  }
   278→  return [{ p: String(v ?? "") }];
   279→}
   280→
   281→function fromEditorValue(key: string, items: Record<string, unknown>[], original: unknown): unknown {
   282→  if (key === "trialCard") {
   283→    const t = (items[0] || {}) as Record<string, string>;
   284→    return { title: t.title || "", sub: t.sub || "", points: (t.points || "").split("\n").map((s) => s.trim()).filter(Boolean) };
   285→  }
   286→  // scalar-string lists (practices, nextSteps, options, topics, story)
   287→  if (Array.isArray(original) && original.length > 0 && typeof original[0] === "string") {
   288→    return items.map((it) => String(it.p ?? it.s ?? it.o ?? it.t ?? "")).filter(Boolean);
   289→  }
   290→  return items.map((it) => {
   291→    const rec: Record<string, unknown> = { ...it };
   292→    for (const k of Object.keys(rec)) {
   293→      if (typeof rec[k] === "string" && (rec[k] as string).includes("\n")) {

   732→function MediaTab({ canEdit, onChanged }: { canEdit: boolean; onChanged: () => void }) {
   733→  const { toast } = useToast();

   735→  const [busy, setBusy] = useState(false);
   736→  const rows = media.data?.media || [];
   737→
   738→  const upload = async (file: File) => {
   739→    setBusy(true);
   740→    try {
   741→      const fd = new FormData();
   742→      fd.append("file", file);
   743→      const res = await fetch("/api/cms/media", { method: "POST", body: fd, credentials: "same-origin" });
   744→      const data = await res.json();
   745→      if (!res.ok) throw new Error(data?.error || "Upload failed");

   747→      media.reload();

   752→  };

   755→    try { await api.del(`/api/cms/media/${m.id}`); media.reload(); onChanged(); toast({ title: "Image deleted", description: m.filename }); }

   771→          </label>
   772→        )}
   773→      </div>
   774→      {rows.length === 0 && !media.loading ? (
   775→        <div className="rounded-2xl border border-dashed border-white/15 bg-brand-deep/30 p-8 text-center">
   776→          <Palette size={22} className="mx-auto text-white/20" />
