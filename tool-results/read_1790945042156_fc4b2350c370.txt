    58→function ListEditor({ spec, value, onChange, canEdit }: {
    59→  spec: Extract<FieldSpec, { type: "list" }>;
    60→  value: Record<string, unknown>[];
    61→  onChange: (next: Record<string, unknown>[]) => void;
    62→  canEdit: boolean;
    63→}) {
    64→  const items = Array.isArray(value) ? value : [];
    65→  const update = (i: number, key: string, v: unknown) =>
    66→    onChange(items.map((it, ix) => (ix === i ? { ...it, [key]: v } : it)));
    67→  const move = (i: number, dir: -1 | 1) => {
    68→    const j = i + dir;
    69→    if (j < 0 || j >= items.length) return;
    70→    const next = [...items];
    71→    [next[i], next[j]] = [next[j], next[i]];
    72→    onChange(next);
    73→  };
    74→  return (
    75→    <div className="space-y-2">
    76→      {items.map((it, i) => (
    77→        <div key={i} className="rounded-xl border border-white/10 bg-brand-deep/50 p-3">
    78→          <div className="mb-2 flex items-center justify-between">
    79→            <span className="text-[10.5px] font-bold uppercase tracking-wider text-white/40">{spec.itemLabel} {i + 1}</span>
    80→            {canEdit && (
    81→              <span className="flex items-center gap-1">
    82→                <button onClick={() => move(i, -1)} className="rounded p-1 text-white/30 hover:bg-white/5 hover:text-white" aria-label="Move up"><ChevronUp size={13} className="rotate-0" /></button>
    83→                <button onClick={() => move(i, 1)} className="rounded p-1 text-white/30 hover:bg-white/5 hover:text-white" aria-label="Move down"><ChevronUp size={13} className="rotate-180" /></button>
    84→                <button onClick={() => onChange(items.filter((_, ix) => ix !== i))} className="rounded p-1 text-white/30 hover:bg-white/5 hover:text-red-300" aria-label="Remove"><X size={13} /></button>
    85→              </span>
    86→            )}
    87→          </div>
    88→          <div className="grid gap-2 sm:grid-cols-2">
    89→            {spec.fields.map((f) => (
    90→              <label key={f.key} className={cn("block", f.type === "area" && "sm:col-span-2")}>
    91→                <span className={lbl}>{f.label}</span>
    92→                {f.type === "area" ? (
    93→                  <textarea value={String(it[f.key] ?? "")} disabled={!canEdit} rows={2}
    94→                    onChange={(e) => update(i, f.key, e.target.value)} className={cn(inpArea, "mt-1")} />
    95→                ) : f.type === "icon" ? (
    96→                  <select value={String(it[f.key] ?? "")} disabled={!canEdit}
    97→                    onChange={(e) => update(i, f.key, e.target.value)} className={cn(inp, "mt-1 appearance-none")}>
    98→                    {ICON_OPTIONS.map((o) => <option key={o} value={o} className="bg-brand">{o}</option>)}
    99→                  </select>
   100→                ) : (
   101→                  <input value={String(it[f.key] ?? "")} disabled={!canEdit}
   102→                    onChange={(e) => update(i, f.key, e.target.value)} className={cn(inp, "mt-1")} />
   103→                )}
   104→              </label>
   105→            ))}
   106→          </div>
   107→        </div>
   108→      ))}
   109→      {canEdit && (
   110→        <Button size="sm" variant="outline" className="h-8 rounded-xl border-white/15 bg-transparent text-[11.5px] text-white/70 hover:bg-white/5 hover:text-white"
   111→          onClick={() => onChange([...items, Object.fromEntries(spec.fields.map((f) => [f.key, ""]))])}>
   112→          <Plus size={12} className="mr-1" /> Add {spec.itemLabel.toLowerCase()}
   113→        </Button>

   115→    </div>

   289→        }

   298→function fromEditorValue(key: string, items: Record<string, unknown>[], original: unknown): unknown {
   299→  if (key === "trialCard") {
   300→    const t = (items[0] || {}) as Record<string, string>;
   301→    return { title: t.title || "", sub: t.sub || "", points: (t.points || "").split("\n").map((s) => s.trim()).filter(Boolean) };
   302→  }
   303→  // scalar-string lists (practices, nextSteps, options, topics, story)
   304→  if (Array.isArray(original) && original.length > 0 && typeof original[0] === "string") {
   305→    return items.map((it) => String(it.p ?? it.s ?? it.o ?? it.t ?? "")).filter(Boolean);
   306→  }
   307→  return items.map((it) => {
   308→    const rec: Record<string, unknown> = { ...it };
   309→    for (const k of Object.keys(rec)) {
   310→      if (typeof rec[k] === "string" && (rec[k] as string).includes("\n")) {
   311→        rec[k] = (rec[k] as string).split("\n").map((s) => s.trimEnd()).filter((s, i, a) => !(i === a.length - 1 && s === ""));
   312→      }
   313→    }
   314→    return rec;
   315→  });
