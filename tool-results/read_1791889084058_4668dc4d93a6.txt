    77→export function roleLabel(role?: string | null): string {
    78→  if (!role) return "Team";
    79→  if (role === "platform_admin") return "Super admin";
    80→  if (role === "platform_manager") return "Manager";
    81→  if (role === "platform_observer") return "Observer";
    82→  if (role === "owner") return "Business owner";
    83→  return role.charAt(0).toUpperCase() + role.slice(1);
    84→}

  1039→function TenantDrawer({ tenant, onClose, onChanged, toast, canOperate, isSuper }: {
  1040→  tenant: TenantVM | null;
  1041→  onClose: () => void;
  1042→  onChanged: () => void;
  1043→  toast: ReturnType<typeof useToast>["toast"];
  1044→  canOperate: boolean;
  1045→  isSuper: boolean;
  1046→}) {
  1047→  const { navigate } = useRouter();
  1048→  const { refresh } = useSession();
  1049→  const open = tenant !== null;
  1050→  const [busy, setBusy] = useState(false);
  1051→  const [planPicking, setPlanPicking] = useState(false);
  1052→  const detail = useApiData<TenantDetail>(tenant ? `/api/admin/tenants/${tenant.id}` : null);
  1053→
  1054→  const patchTenant = async (data: { status?: string; planId?: string }, label: string) => {
  1055→    if (!tenant) return;
  1056→    setBusy(true);
  1057→    try {
  1058→      await api.patch(`/api/admin/tenants/${tenant.id}`, data);
  1059→      toast({ title: label, description: `${tenant.business} updated.` });
  1060→      detail.reload();
  1061→      onChanged();
  1062→    } catch (e) {
  1063→      toast({ title: "Update failed", description: e instanceof Error ? e.message : "", variant: "destructive" });
  1064→    } finally {
  1065→      setBusy(false);
  1066→    }
  1067→  };
  1068→
  1069→  const impersonate = async () => {
  1070→    if (!tenant) return;
  1071→    setBusy(true);
  1072→    try {
  1073→      await api.post(`/api/admin/tenants/${tenant.id}/impersonate`);
  1074→      await refresh();
  1075→      toast({ title: "Impersonation started", description: `Time-boxed 30-min session into ${tenant.business}. Audit-logged.` });
  1076→      onClose();
  1077→      navigate({ surface: "client" });
  1078→    } catch (e) {
  1079→      toast({ title: "Could not start session", description: e instanceof Error ? e.message : "", variant: "destructive" });
  1080→    } finally {
  1081→      setBusy(false);
  1082→    }
  1083→  };
  1084→
  1085→  const t = detail.data?.tenant || tenant;
  1086→  const d = detail.data?.detail;
  1087→
  1088→  return (
  1089→    <div className={cn("fixed inset-0 z-50", !open && "pointer-events-none")} aria-hidden={!open}>
  1090→      <button
  1091→        onClick={onClose}
  1092→        aria-label="Close client details"
  1093→        tabIndex={open ? 0 : -1}
  1094→        className={cn("absolute inset-0 bg-black/60 backdrop-blur-sm transition-opacity duration-300", open ? "opacity-100" : "opacity-0")}
  1095→      />
  1096→      <aside
  1097→        className={cn(
  1098→          "absolute right-0 top-0 flex h-full w-full max-w-md flex-col border-l border-white/10 bg-brand shadow-2xl transition-transform duration-300 ease-out",
  1099→          open ? "translate-x-0" : "translate-x-full"
  1100→        )}
  1101→        role="dialog"
  1102→        aria-label="Client details"
  1103→      >
  1104→        {t && d && (
  1105→          <TenantDrawerBody
  1106→            tenant={t}
  1107→            detail={d}
  1108→            busy={busy}
  1109→            planPicking={planPicking}
  1110→            setPlanPicking={setPlanPicking}
  1111→            onPatch={patchTenant}
  1112→            onImpersonate={impersonate}
  1113→            onClose={onClose}
  1114→            toast={toast}
  1115→            canOperate={canOperate}
  1116→            isSuper={isSuper}
  1117→          />
  1118→        )}
  1119→        {tenant && detail.loading && (
  1120→          <div className="flex flex-1 items-center justify-center">
  1121→            <RefreshCw size={22} className="animate-spin text-mint" />
  1122→          </div>
  1123→        )}
  1124→      </aside>
  1125→    </div>
  1126→  );
  1127→}

  1129→function TenantDrawerBody({ tenant: t, detail: d, busy, planPicking, setPlanPicking, onPatch, onImpersonate, onClose, toast, canOperate, isSuper }: {

  1569→// Personal profile + password for platform staff (super admin, managers, observers)
  1570→function AccountView({ onRefresh }: { onRefresh: () => Promise<unknown> }) {
  1571→  const { toast } = useToast();
  1572→  const { user } = useSession();
  1573→  const account = useApiData<AccountPayload>("/api/account");
  1574→  const [profile, setProfile] = useState({ name: "", email: "", phone: "" });
  1575→  const [pw, setPw] = useState({ current: "", next: "", confirm: "" });
  1576→  const [savingProfile, setSavingProfile] = useState(false);
  1577→  const [savingPw, setSavingPw] = useState(false);
  1578→  const [loaded, setLoaded] = useState(false);
  1579→
  1580→  useEffect(() => {
  1581→    if (account.data && !loaded) {
  1582→      const a = account.data.account;
  1583→      setProfile({ name: a.name, email: a.email, phone: a.phone });
  1584→      setLoaded(true);

  1587→
  1588→  const saveProfile = async () => {
  1589→    setSavingProfile(true);
  1590→    try {
  1591→      await api.patch("/api/account", profile);
  1592→      await onRefresh();
  1593→      account.reload();

  1595→    } catch (e) {

  1601→

  1604→      toast({ title: "Password too short", description: "Use at least 8 characters.", variant: "destructive" });
  1605→      return;

  1613→      await api.post("/api/account/password", { currentPassword: pw.current, newPassword: pw.next });
