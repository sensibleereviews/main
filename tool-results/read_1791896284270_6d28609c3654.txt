   369→function AddLocationInline({ onAdded }: { onAdded: () => void }) {
   370→  const { toast } = useToast();
   371→  const [open, setOpen] = useState(false);
   372→  const [form, setForm] = useState({ name: "", area: "" });
   373→  const [busy, setBusy] = useState(false);
   374→  const add = async () => {
   375→    if (!form.name.trim()) return;
   376→    setBusy(true);
   377→    try {
   378→      await api.post("/api/locations", form);
   379→      toast({ title: "Profile connected", description: `${form.name} will sync within 15 minutes.` });
   380→      setForm({ name: "", area: "" });
   381→      setOpen(false);
   382→      onAdded();
   383→    } catch (e) {
   384→      toast({ title: "Could not connect", description: e instanceof Error ? e.message : "", variant: "destructive" });
   385→    } finally {
   386→      setBusy(false);
   387→    }
   388→  };
   389→  if (!open) {
   390→    return (
   391→      <button onClick={() => setOpen(true)} className="flex w-full items-center justify-center gap-1.5 rounded-xl border border-dashed border-neutral-300 py-3 text-[12px] font-semibold text-neutral-400 transition-colors hover:border-mint/60 hover:text-mint-deep">
   392→        <Plus size={13} /> Connect another profile
   393→      </button>
   394→    );
